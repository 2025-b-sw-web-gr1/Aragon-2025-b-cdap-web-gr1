🏷️ DECORADORES EN ENDPOINTS - TEAMS

  ✅ POST /teams
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiResponse({ status: 201, type: Team })
     - @ApiBadRequestResponse()

  ✅ GET /teams
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiResponse({ status: 200, type: [Team] })

  ✅ GET /teams/:id
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiParam({ name: 'id', example: 1 })
     - @ApiResponse({ status: 200, type: Team })
     - @ApiNotFoundResponse()

  ✅ PUT /teams/:id
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiParam({ name: 'id' })
     - @ApiResponse({ status: 200, type: Team })
     - @ApiBadRequestResponse()
     - @ApiNotFoundResponse()

  ✅ DELETE /teams/:id
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiParam({ name: 'id' })
     - @ApiResponse({ status: 200 })
     - @ApiNotFoundResponse()

  ✅ GET /teams/:id/players
     - @ApiTags('teams')
     - @ApiOperation({ summary, description })
     - @ApiParam({ name: 'id' })
     - @ApiResponse({ status: 200, type: [Player] })
     - @ApiNotFoundResponse()
